\section*{Bài I (2,0 điểm)}
Cho hai biểu thức
\[
A=\frac{3\sqrt{x}}{\sqrt{x}+2},\qquad
B=\frac{x+4}{x-4}-\frac{2}{\sqrt{x}-2},
\]
với $x\ge0$, $x\ne4$.
\begin{enumerate}
  \item Tính giá trị của $A$ khi $x=9$.
  \item Chứng minh rằng $B=\dfrac{\sqrt{x}}{\sqrt{x}+2}$.
  \item Tìm số nguyên dương $x$ lớn nhất thỏa mãn $A-B<\dfrac32$.
\end{enumerate}

\section*{Bài II (2,0 điểm)}
\begin{enumerate}
  \item Giải bài toán bằng cách lập phương trình hoặc hệ phương trình: Một ô tô và một xe máy cùng khởi hành từ địa điểm $A$ đi đến địa điểm $B$. Vận tốc ô tô lớn hơn vận tốc xe máy $20\,\mathrm{km/h}$ nên ô tô đến $B$ sớm hơn xe máy 30 phút. Biết quãng đường $AB$ dài $60\,\mathrm{km}$, tính vận tốc mỗi xe. Giả định vận tốc mỗi xe không đổi trên toàn bộ quãng đường.
  \item Quả bóng đá dùng trong các trận thi đấu dành cho trẻ em từ 6 đến 8 tuổi có dạng hình cầu bán kính $9{,}5\,\mathrm{cm}$. Tính diện tích bề mặt quả bóng (lấy $\pi\approx3{,}14$).
\end{enumerate}

\section*{Bài III (2,5 điểm)}
\begin{enumerate}
  \item Giải hệ phương trình
  \[
  \begin{cases}
  2x+\dfrac{12}{y+2}=5,\\[3pt]
  3x-\dfrac{4}{y+2}=2.
  \end{cases}
  \]
  \item Trong mặt phẳng tọa độ $Oxy$, cho parabol $(P):y=x^2$ và đường thẳng $(d):y=2x+m^2$.
  \begin{enumerate}[label=\alph*)]
    \item Chứng minh $(d)$ luôn cắt $(P)$ tại hai điểm phân biệt.
    \item Tìm tất cả giá trị của $m$ để $(d)$ cắt $(P)$ tại hai điểm phân biệt có hoành độ $x_1,x_2$ thỏa mãn $(x_1+1)(x_2+1)=-3$.
  \end{enumerate}
\end{enumerate}

\section*{Bài V (0,5 điểm)}
Với các số thực không âm $x,y$ thỏa mãn $x^2+y^2=4$, tìm giá trị nhỏ nhất của
\[
P=x+2y.
\]
