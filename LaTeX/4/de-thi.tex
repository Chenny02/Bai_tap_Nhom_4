\section*{Câu I (2,0 điểm)}
Cho hai biểu thức
\[
A=\frac{x+2}{\sqrt{x}},\qquad
B=\frac{2\sqrt{x}-3}{\sqrt{x}-1}+\frac{3-\sqrt{x}}{x-1},
\]
với $x>0$, $x\ne 1$.
\begin{enumerate}
  \item Tính giá trị của biểu thức $A$ khi $x=9$.
  \item Chứng minh rằng $B=\dfrac{2\sqrt{x}}{\sqrt{x}+1}$.
  \item Tìm tất cả các giá trị của $x$ để $A\cdot B=4$.
\end{enumerate}

\section*{Câu II (2,0 điểm)}
\begin{enumerate}
  \item Giải bài toán bằng cách lập phương trình hoặc hệ phương trình: Theo kế hoạch, một phân xưởng phải làm xong 900 sản phẩm trong một số ngày quy định. Thực tế, mỗi ngày phân xưởng làm được nhiều hơn 15 sản phẩm so với số sản phẩm phải làm trong một ngày theo kế hoạch. Vì thế, ba ngày trước khi hết thời hạn, phân xưởng đã làm xong 900 sản phẩm. Hỏi theo kế hoạch, mỗi ngày phân xưởng phải làm bao nhiêu sản phẩm? Giả định số sản phẩm phân xưởng làm được trong mỗi ngày là bằng nhau.
  \item Một khối gỗ dạng hình trụ có bán kính đáy $30\,\mathrm{cm}$ và chiều cao $120\,\mathrm{cm}$. Tính thể tích khối gỗ đó (lấy $\pi\approx3{,}14$).
\end{enumerate}

\section*{Câu III (2,5 điểm)}
\begin{enumerate}
  \item Giải hệ phương trình
  \[
  \begin{cases}
  \dfrac{2}{x-3}-3y=1,\\[3pt]
  \dfrac{3}{x-3}+2y=8.
  \end{cases}
  \]
  \item Trong mặt phẳng tọa độ $Oxy$, cho parabol $(P):y=x^2$ và đường thẳng $(d):y=(m+2)x-m$.
  \begin{enumerate}[label=\alph*)]
    \item Chứng minh $(d)$ luôn cắt $(P)$ tại hai điểm phân biệt.
    \item Gọi $x_1,x_2$ là hoành độ các giao điểm của $(d)$ và $(P)$. Tìm tất cả giá trị của $m$ để
    \[
    \frac1{x_1}+\frac1{x_2}=\frac1{x_1+x_2-2}.
    \]
  \end{enumerate}
\end{enumerate}

\section*{Câu V (0,5 điểm)}
Cho hai số thực dương $a,b$ thỏa mãn $a+b\le2$. Chứng minh rằng
\[
\frac{a^2}{a^2+b}+\frac{b^2}{b^2+a}\le1.
\]
